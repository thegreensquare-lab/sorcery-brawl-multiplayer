```cpp #include const int lightPin = A1; const int servoPin = 11; Servo myServo; int lightValue; int servoPosition; void setup() { pinMode(lightPin, INPUT); Serial.begin(9600); myServo.attach(servoPin); myServo.write(0); } void loop() { lightValue = analogRead(lightPin); // Convert sensor readings from 300-900 into angles from 0-180 servoPosition = map(lightValue, 300, 900, 0, 180); // Keep the servo angle between 0 and 180 degrees servoPosition = constrain(servoPosition, 0, 180); // Move the servo to the corresponding angle myServo.write(servoPosition); // Display the sensor reading and servo angle Serial.print("Light Reading: "); Serial.print(lightValue); Serial.print(" | Servo Angle: "); Serial.println(servoPosition); delay(200); } ```
